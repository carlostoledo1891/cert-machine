"""Sequential-LP trust-region descent for C_1d upper bound: minimize 2n*max|v*v|/(sum v)^2.
Usage: slp_1d.py in.json out.json [upsample_factor] [iters]"""
import json, sys, time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def score(v):
    return 2 * len(v) * np.max(np.abs(np.convolve(v, v))) / v.sum() ** 2

def conv_matrix(v):
    n = len(v)
    col = np.concatenate([v, np.zeros(n - 1)]); row = np.zeros(n); row[0] = v[0]
    return toeplitz(col, row)  # (2n-1) x n, (M d)_m = sum_i v_{m-i} d_i

d = json.load(open(sys.argv[1])); v = np.array(d["values"], float)
up = int(sys.argv[3]) if len(sys.argv) > 3 else 1
iters = int(sys.argv[4]) if len(sys.argv) > 4 else 200
v = np.repeat(v, up); n = len(v)
v = v * (2 * n / v.sum())          # S = 2n  => score = max|c| / (2n)
print("start n", n, "score", score(v), flush=True)
rho = 0.02 * np.abs(v).mean(); t0 = time.time()
for it in range(iters):
    c = np.convolve(v, v); M = 2 * conv_matrix(v)
    # vars: d (n), t ; min t ; c + M d <= t ; -(c + M d) <= t ; sum d = 0 ; |d|<=rho
    A = np.block([[M, -np.ones((2*n-1, 1))], [-M, -np.ones((2*n-1, 1))]])
    b = np.concatenate([-c, c])
    res = linprog(np.r_[np.zeros(n), 1.0], A_ub=A, b_ub=b, A_eq=np.r_[np.ones(n), 0][None], b_eq=[0],
                  bounds=[(-rho, rho)] * n + [(None, None)], method="highs")
    if res.status != 0: print("lp fail", res.message); break
    dv = res.x[:n]; old = score(v)
    for _ in range(8):  # backtrack on true objective
        nv = v + dv; nv *= 2 * n / nv.sum()
        if score(nv) < old: break
        dv *= 0.5
    if score(nv) < old:
        v = nv; rho = min(rho * 1.5, 0.2 * np.abs(v).mean())
    else:
        rho *= 0.3
    if it % 10 == 0 or it == iters - 1:
        print(f"it {it} score {score(v):.12f} rho {rho:.2e} t {time.time()-t0:.0f}s", flush=True)
    if rho < 1e-12: break
json.dump({"values": v.tolist(), "float_score": score(v)}, open(sys.argv[2], "w"))
print("final", score(v))
