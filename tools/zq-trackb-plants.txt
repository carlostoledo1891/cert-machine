-- ZQ-PLANT Q1 extra-hypothesis
import CondZQ
import Frozen.BoundedIntegerPolynomial

/-- an extra first hypothesis, `1 = 0`, and the frozen statement from it by `absurd` -/
theorem CertMachineZQPlant.q1_extra_hypothesis (h0 : (1 : Nat) = 0) (hM : CondZQ.HypM) (hH : CondZQ.HypH)
    (hPC : CondZQ.HypPC) (hC : CondZQ.HypC) (hLT : CondZQ.HypLT) (hCC : CondZQ.HypCC) (hH1 : CondZQ.HypH1)
    (hH2 : CondZQ.HypH2) :
    type_of% @ZInQ.exists_bounded_integer_polynomial_defining_integers :=
  absurd h0 (by decide)
-- ZQ-PLANT Q2 circular-hypothesis

/-- HypH2 replaced by HypCirc, and the frozen statement from it -/

    (hC : CondZQ.HypC) (hLT : CondZQ.HypLT) (hCC : CondZQ.HypCC) (hH1 : CondZQ.HypH1)
